// Dynamic Platform Parser & Button Generator function parseTrackSource(url) { const cleanUrl = url.split('?')[0]; // 1. SoundCloud if (cleanUrl.includes('soundcloud.com')) { return { type: 'iframe', embedUrl: `https://w.soundcloud.com/player/?url=${encodeURIComponent(cleanUrl)}&color=%238b5cf6&auto_play=true`, btnText: '☁️ Follow on SoundCloud', actionUrl: cleanUrl }; } // 2. Bandcamp if (cleanUrl.includes('bandcamp.com')) { return { type: 'iframe', // Bandcamp vereist de standaard pagina link voor hun embed embedUrl: cleanUrl, btnText: '🛒 Buy / Support on Bandcamp', actionUrl: cleanUrl }; } // 3. YouTube / Shorts if (cleanUrl.includes('youtube.com') || cleanUrl.includes('youtu.be')) { const ytId = cleanUrl.match(/(?:youtu\.be\/|youtube\.com\/(?:watch\?v=|shorts\/))([\w-]{11})/)[1]; return { type: 'iframe', embedUrl: `https://www.youtube.com/embed/${ytId}?autoplay=1`, btnText: '🔔 Subscribe on YouTube', actionUrl: cleanUrl }; } // 4. Direct MP3 File if (cleanUrl.endsWith('.mp3') || cleanUrl.endsWith('.wav')) { return { type: 'audio', embedUrl: cleanUrl, btnText: '📥 Download / Support Track', actionUrl: cleanUrl }; } // Fallback (Generieke Link) return { type: 'iframe', embedUrl: cleanUrl, btnText: '🔗 Open Original Platform', actionUrl: cleanUrl }; } // Speler Laden & Knop Dynamisch Aanpassen function renderMedia(trackUrl) { const source = parseTrackSource(trackUrl); const container = document.getElementById('mediaContainer'); const platformBtn = document.getElementById('platformFollowBtn'); if (source.type === 'iframe') { container.innerHTML = ``; } else if (source.type === 'audio') { container.innerHTML = ``; } // Pas de knoptekst en URL dynamisch aan voor het ontgrendelde scherm platformBtn.innerText = source.btnText; platformBtn.href = source.actionUrl; }
